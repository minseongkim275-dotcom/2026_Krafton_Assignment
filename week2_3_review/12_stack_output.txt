=== 괄호 검사 ===
'{[()]}' -> True
'()[]{}' -> True
'([)]' -> False
'(' -> False
')' -> False
'' -> True
'a(b[c]d)e' -> True
'(()' -> False

=== 후위 표기식 계산 ===
2 3 + 4 * = 20
5 1 2 + 4 * + 3 - = 14
7 2 / = 3
-7 2 / = -3
42 = 42

=== 오른쪽에서 처음 나오는 더 큰 수 ===
[2, 1, 2, 4, 3] -> [4, 2, 4, -1, -1]
[1, 2, 3, 4] -> [2, 3, 4, -1]
[4, 3, 2, 1] -> [-1, -1, -1, -1]
[5, 5, 5] -> [-1, -1, -1]
[9] -> [-1]
[] -> []
