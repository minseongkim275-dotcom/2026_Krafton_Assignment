=== 두 수의 합 ===
[2, 7, 11, 15], target=9 -> (0, 1)
[3, 2, 4], target=6 -> (1, 2)
[3, 3], target=6 -> (0, 1)
[1, 2, 3], target=100 -> None

  ['ate', 'eat', 'tea']
  ['bat']
  ['nat', 'tan']
